cmake_minimum_required(VERSION 3.25)
project(holonight-viewer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)
set(HOLONIGHT_DEPENDENCY_PREFIX "${CMAKE_SOURCE_DIR}/build/deps/prefix" CACHE PATH "Installed provider prefix")
list(PREPEND CMAKE_PREFIX_PATH "${HOLONIGHT_DEPENDENCY_PREFIX}")
set(HOLONIGHT_QML_IMPORT_PATH "${HOLONIGHT_DEPENDENCY_PREFIX}/${CMAKE_INSTALL_LIBDIR}/qt6/qml" CACHE PATH "Installed provider QML modules")
set(QML_IMPORT_PATH "${HOLONIGHT_QML_IMPORT_PATH}" CACHE STRING "Development QML imports" FORCE)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Quick Qml QuickControls2)
find_package(HolonightQt CONFIG REQUIRED)
qt_standard_project_setup(REQUIRES 6.11)
option(BUILD_TESTING "Build smoke tests" OFF)
add_subdirectory(apps/viewer)
if(BUILD_TESTING)
  enable_testing()
  find_package(Qt6 6.11 REQUIRED COMPONENTS Test)
  find_package(GTest CONFIG REQUIRED)
  add_subdirectory(tests)
endif()
install(FILES LICENSE LICENSES/GPL-3.0-or-later.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/${PROJECT_NAME})
install(FILES packaging/org.holonight.Viewer.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/org.holonight.Viewer.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
find_program(CLANG_FORMAT clang-format)
find_program(RUN_CLANG_TIDY run-clang-tidy)
file(GLOB VIEWER_CXX CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/apps/viewer/*.cpp" "${CMAKE_SOURCE_DIR}/tests/*.cpp")
file(GLOB VIEWER_FORMAT_HEADERS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/apps/viewer/*.h")
set(VIEWER_FORMAT_FILES ${VIEWER_CXX} ${VIEWER_FORMAT_HEADERS})
if(CLANG_FORMAT)
  add_custom_target(format COMMAND "${CLANG_FORMAT}" -i ${VIEWER_FORMAT_FILES} VERBATIM)
  add_custom_target(format-check COMMAND "${CLANG_FORMAT}" --dry-run --Werror ${VIEWER_FORMAT_FILES} VERBATIM)
endif()
if(RUN_CLANG_TIDY)
  add_custom_target(tidy COMMAND ${RUN_CLANG_TIDY} -p ${CMAKE_BINARY_DIR} -removed-arg=-mno-direct-extern-access ${VIEWER_CXX})
endif()
add_custom_target(qml-lint DEPENDS viewer-ui_qmllint)
