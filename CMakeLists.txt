cmake_minimum_required(VERSION 3.25)
project(holonight-viewer VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(GNUInstallDirs)
# Provider discovery belongs to the caller; presets supply development defaults.
set(VIEWER_RUNTIME_QML_PATH "${QML_IMPORT_PATH}")
string(REPLACE ";" ":" VIEWER_RUNTIME_QML_PATH "${VIEWER_RUNTIME_QML_PATH}")
function(viewer_warnings target)
  target_compile_options(${target} PRIVATE
    $<$<COMPILE_LANG_AND_ID:CXX,GNU,Clang>:-Wall;-Wextra;-Wpedantic>
    $<$<COMPILE_LANG_AND_ID:CXX,MSVC>:/W4>)
endfunction()
set(QT_NO_PRIVATE_MODULE_WARNING ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui GuiPrivate DBus Quick Qml QuickControls2)
find_package(HolonightQt CONFIG REQUIRED)
qt_standard_project_setup(REQUIRES 6.11)
option(BUILD_TESTING "Build smoke tests" OFF)
add_subdirectory(apps/viewer)
if(BUILD_TESTING)
  enable_testing()
  find_package(Qt6 6.11 REQUIRED COMPONENTS Test)
  find_package(GTest CONFIG REQUIRED)
  add_subdirectory(tests)
endif()
install(FILES LICENSE LICENSES/GPL-3.0-or-later.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/${PROJECT_NAME})
install(FILES packaging/org.holonight.Viewer.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/org.holonight.Viewer.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
find_program(RUN_CLANG_TIDY run-clang-tidy)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(GLOB_RECURSE VIEWER_CXX CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/apps/*.cpp" "${CMAKE_SOURCE_DIR}/tests/*.cpp")
add_custom_target(format COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/format-sources.py" VERBATIM)
add_custom_target(format-check COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/format-sources.py" --check VERBATIM)
add_custom_target(qml-import-check COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/check-qml-import-policy.py" VERBATIM)
add_custom_target(qmltypes-check COMMAND bash "${CMAKE_SOURCE_DIR}/scripts/check-qmltypes.sh" "${CMAKE_BINARY_DIR}" DEPENDS holonight-viewer VERBATIM)
if(RUN_CLANG_TIDY)
  add_custom_target(tidy COMMAND ${RUN_CLANG_TIDY} -p ${CMAKE_BINARY_DIR} -removed-arg=-mno-direct-extern-access -config-file=${CMAKE_SOURCE_DIR}/.clang-tidy -j 2 ${VIEWER_CXX})
endif()
add_custom_target(qml-lint DEPENDS holonight-viewer_qmllint)
